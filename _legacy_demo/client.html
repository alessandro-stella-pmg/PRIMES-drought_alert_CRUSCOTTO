const notificationsContainer = document.getElementById('notifications'); const wsUrl = `ws://${window.location.hostname}:${window.location.port}`; const socket = new WebSocket(wsUrl); socket.addEventListener('open', () => { statusDot.className = 'status-dot bg-success'; statusText.innerText = 'Connected'; document.getElementById('last-event-text').innerText = 'Connected to backend. Waiting for notifications...'; console.log('[client] WebSocket connected'); }); socket.addEventListener('message', event => { console.log('[client] WebSocket message', event.data); const payload = JSON.parse(event.data); if (payload.event === 'notification') { addNotificationCard(payload.data); } }); socket.addEventListener('close', () => { statusDot.className = 'status-dot bg-danger'; statusText.innerText = 'Disconnected'; }); socket.addEventListener('error', () => { statusDot.className = 'status-dot bg-danger'; statusText.innerText = 'Connection error'; }); function addNotificationCard(data) { const card = document.createElement('div'); card.className = 'col-12'; document.getElementById('last-event-text').innerText = `Last received: ${escapeHtml(data.title)}`; card.innerHTML = `
${escapeHtml(data.title)}
${escapeHtml(data.type)} • ${new Date(data.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
${escapeHtml(data.type)}

${escapeHtml(data.message)}

Target: ${escapeHtml(data.target || 'all')}
`; notificationsContainer.prepend(card); if (window.Notification && Notification.permission === 'granted') { new Notification(data.title, { body: data.message }); } else if (window.Notification && Notification.permission !== 'denied') { Notification.requestPermission(); } } function escapeHtml(text) { return String(text) .replace(/&/g, '&') .replace(//g, '>') .replace(/"/g, '"') .replace(/'/g, '''); }
